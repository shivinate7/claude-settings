test('every file that shows a Thousands cell also carries the marker', () => {
  const files = walk('src/components').filter((f) => f.endsWith('.tsx') && !f.endsWith('Thousands.tsx'));
  const users = files.filter((f) => /<Thousands\s/.test(readFileSync(f, 'utf8')));
  assert.ok(users.length >= 10, `expected the money screens to use Thousands, found ${users.length}`);
  const bare = users.filter((f) => !/<ThousandsNote\s*\/>|IN_THOUSANDS/.test(readFileSync(f, 'utf8')));
  assert.deepEqual(bare, [], 'a file shows amounts in thousands with no marker');
});

  const src = readFileSync('src/components/Dashboard.tsx', 'utf8').replace(/\/\*[\s\S]*?\*\/|\/\/[^\n]*/g, '');

  // panel. The note sat above the panel and repeated the head (hub ruling, class c, 2026-10-12).
  const card = src.slice(src.indexOf('const breakdownCard = ('), src.indexOf('const hostZones = ('));
  assert.ok(card.length > 0, 'the breakdown card is not found in Dashboard.tsx, so nothing was read');
  assert.match(card, /\{tabView\}/, 'the breakdown card does not render the review panel that holds the Cost head');

test('the Compare lines pane has no note above its list, and its one unit marker sits in the money head', () => {
  // D-632, the owner's ruling of 2026-10-11, and the hub ruling on round 2: the note goes, and the pane's money
  // head line holds the one marker. A match anywhere in the file is not a match in the right place, so this reads
  // the lines pane's own block: from its `const linesPane` to the next sibling `const`, with comments removed.
  const src = stripComments(readFileSync('src/components/screens/ComparisonScreen.tsx', 'utf8'), 'src/components/screens/ComparisonScreen.tsx');
  const start = src.indexOf('const linesPane = ');
  assert.ok(start >= 0, 'the lines pane is not found in ComparisonScreen.tsx, so nothing was read');
  const end = src.indexOf('\n  const ', start + 1);
  const pane = src.slice(start, end);
  const list = pane.indexOf('<ul');
  assert.ok(list > 0, 'the lines pane block is not the lines pane: no list in it, so nothing was read');
  assert.doesNotMatch(pane, /<ThousandsNote\b/, 'the lines pane renders a ThousandsNote paragraph above its list');
  // Exactly one marker in the pane. A second one, in a cell or a note, fails here.
  const markers = pane.match(/IN_THOUSANDS|\(in thousands\)/g) ?? [];
  assert.equal(markers.length, 1, `the lines pane should carry one unit marker, in its money head, and it carries ${markers.length}`);
  // The head is the element directly above the amounts. From the marker to the list, the source holds only the
  // marker's own text, the braces that close its expression, and closing tags. So no sentence, in either view,
  // sits between the money head and its amounts (the series sentence fails this, in the series view and the codes view).
  const head = pane.slice(0, list);
  const markerAt = head.search(/IN_THOUSANDS|\(in thousands\)/);
  assert.ok(markerAt >= 0, 'the one unit marker sits below the list, not in the money head above it');
  assert.match(head.slice(markerAt), /^[^<]*(<\/[A-Za-z]+>\s*)+$/,
    'a sentence or an element sits between the money head and the amounts below it. The head must be the element directly above the list');
  const line = head.slice(head.lastIndexOf('\n', markerAt) + 1, head.indexOf('\n', markerAt) === -1 ? head.length : head.indexOf('\n', markerAt));
  assert.doesNotMatch(line, /<p\b/, 'the unit marker sits in a paragraph above the list, which is a note, not a money head');
});

test('a ThousandsNote is never followed by a table in the chat cards and the Level pane', () => {
  // D-632, the owner's ruling of 2026-10-11: the unit sits in a money column head, so a note above a table is a finding.
  // These three files draw their tables in chat and in the Level pane, which the browser tier does not reach. So this
  // reads the source: each ThousandsNote, with comments removed, must have no DataTable or table after it before the
  // next note. A note followed by a table is the note the lane removes.
  for (const file of ['src/components/PreconCard.tsx', 'src/components/LinesCard.tsx', 'src/components/ItbBidCompare.tsx']) {
    const src = stripComments(readFileSync(file, 'utf8'), file);
    const notes = [...src.matchAll(/<ThousandsNote\s*\/>/g)].map((m) => m.index);
    const found = notes.filter((at, i) => {
      const next = i + 1 < notes.length ? notes[i + 1] : src.length;
      return /<(DataTable|table)\b/.test(src.slice(at, next));
    }).map((at) => src.slice(0, at).split('\n').length);
    assert.deepEqual(found, [], `${file}: a ThousandsNote sits above a table at line(s) ${found.join(', ')}. Move the unit into the table's money head`);
  }
});

  const bad = [];
  for (const file of walk('src/components').filter((f) => f.endsWith('.tsx'))) {
    const src = readFileSync(file, 'utf8');
