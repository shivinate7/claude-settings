test('the Compare lines pane has no note above its list, and its one unit marker sits in the money head', () => {
  // D-632, the owner's ruling of 2026-10-11, and the hub ruling on round 2: the note goes, and the pane's money
  // head line holds the one marker. A match anywhere in the file is not a match in the right place, so this reads
  // the lines pane's own block: from its `const linesPane` to the next sibling `const`, with comments removed.
  const src = stripComments(readFileSync('src/components/screens/ComparisonScreen.tsx', 'utf8'), 'src/components/screens/ComparisonScreen.tsx');
  const start = src.indexOf('const linesPane = ');
  assert.ok(start >= 0, 'the lines pane is not found in ComparisonScreen.tsx, so nothing was read');
  const end = src.indexOf('\n  const ', start + 1);
  const pane = src.slice(start, end);
  const list = pane.indexOf('<ul');
  assert.ok(list > 0, 'the lines pane block is not the lines pane: no list in it, so nothing was read');
  assert.doesNotMatch(pane, /<ThousandsNote\b/, 'the lines pane renders a ThousandsNote paragraph above its list');
  // Exactly one marker in the pane. A second one, in a cell or a note, fails here.
  const markers = pane.match(/IN_THOUSANDS|\(in thousands\)/g) ?? [];
  assert.equal(markers.length, 1, `the lines pane should carry one unit marker, in its money head, and it carries ${markers.length}`);
  // The head is the element directly above the amounts. From the marker to the list, the source holds only the
  // marker's own text, the braces that close its expression, and closing tags. So no sentence, in either view,
  // sits between the money head and its amounts (the series sentence fails this, in the series view and the codes view).
  const head = pane.slice(0, list);
  const markerAt = head.search(/IN_THOUSANDS|\(in thousands\)/);
  assert.ok(markerAt >= 0, 'the one unit marker sits below the list, not in the money head above it');
  assert.match(head.slice(markerAt), /^[^<]*(<\/[A-Za-z]+>\s*)+$/,
    'a sentence or an element sits between the money head and the amounts below it. The head must be the element directly above the list');
  const line = head.slice(head.lastIndexOf('\n', markerAt) + 1, head.indexOf('\n', markerAt) === -1 ? head.length : head.indexOf('\n', markerAt));
  assert.doesNotMatch(line, /<p\b/, 'the unit marker sits in a paragraph above the list, which is a note, not a money head');
});

test('every table whose cells render Thousands has exactly one money head that carries the unit', () => {
  // D-632, the owner's ruling of 2026-10-11: one unit marker for each table, in its money column head. A table
  // of Thousands cells is read through the TypeScript parser, not by text match: each DataTable (or table) that
  // holds a <Thousands> cell must have exactly one head that names IN_THOUSANDS or "(in thousands)".
  // Heads are read from the `columns` expression, through `names(...)`, a const in the same file, and `?:` / `&&`.
  const bad = [];
  for (const file of walk('src/components').filter((f) => f.endsWith('.tsx'))) {
    const src = readFileSync(file, 'utf8');
    const sf = ts.createSourceFile(file, src, ts.ScriptTarget.Latest, true, ts.ScriptKind.TSX);
    const consts = new Map();
    const collectConsts = (node) => {
      if (ts.isVariableDeclaration(node) && ts.isIdentifier(node.name) && node.initializer) consts.set(node.name.text, node.initializer);
      ts.forEachChild(node, collectConsts);
    };
    collectConsts(sf);
    const containsThousands = (node) => {
      let found = false;
      const visit = (n) => {
        if (found) return;
        if ((ts.isJsxSelfClosingElement(n) || ts.isJsxOpeningElement(n)) && n.tagName.getText() === 'Thousands') { found = true; return; }
        ts.forEachChild(n, visit);
      };
      visit(node);
      return found;
    };
    // Head texts of a `columns` expression. An unreadable part is recorded as null, so it fails the check.
    const headsOf = (expr) => {
      if (ts.isParenthesizedExpression(expr) || ts.isAsExpression(expr)) return headsOf(expr.expression);
      if (ts.isIdentifier(expr)) return consts.has(expr.text) ? headsOf(consts.get(expr.text)) : [null];
      if (ts.isCallExpression(expr)) return expr.arguments.length > 0 ? headsOf(expr.arguments[0]) : [null];
      if (ts.isConditionalExpression(expr)) return [...headsOf(expr.whenTrue), ...headsOf(expr.whenFalse)];
      if (ts.isBinaryExpression(expr) && expr.operatorToken.kind === ts.SyntaxKind.AmpersandAmpersandToken) return headsOf(expr.right);
      if (ts.isArrayLiteralExpression(expr)) {
        return expr.elements.flatMap((el) => {
          if (ts.isSpreadElement(el)) return headsOf(el.expression);
          if (!ts.isObjectLiteralExpression(el)) return [null];
          const head = el.properties.find((p) => ts.isPropertyAssignment(p) && p.name.getText() === 'head');
          return head ? [head.initializer.getText()] : [null];
        });
      }
      return [null];
    };
    const line = (node) => sf.getLineAndCharacterOfPosition(node.getStart()).line + 1;
    // The unit in a head: the thousands marker, or Compare's "(K)" money head (`moneyHead`), which marks a money column.
    const UNIT_IN_HEAD = /IN_THOUSANDS|\(in thousands\)|moneyHead\(/g;
    const check = (node, label, heads, columnsText) => {
      if (heads.includes(null)) {
        // A head the parser cannot read (a head built in a map or an import): count the marker in the columns text.
        const marked = (columnsText.match(UNIT_IN_HEAD) ?? []).length;
        if (marked !== 1) bad.push(`${file}:${line(node)} ${label}: a head cannot be read, and its columns carry ${marked} unit mark(s), need exactly one`);
        return;
      }
      const marked = heads.filter((h) => /IN_THOUSANDS|\(in thousands\)|moneyHead\(/.test(h)).length;
      if (marked !== 1) { bad.push(`${file}:${line(node)} ${label}: ${marked} money head(s) carry the unit, need exactly one. Heads: ${heads.join(' | ')}`); return; }
      // Head over money: the unit head is the column whose cells hold a Thousands figure. A head over a vendor or a
      // name column (its cells hold no amount) states the unit over the wrong column.
      const at = heads.findIndex((h) => /IN_THOUSANDS|\(in thousands\)|moneyHead\(/.test(h));
      const money = moneyColumnsOf(node);
      // A column list with a spread of a condition (`...(on ? [...] : [])`) has columns that may not render, so the head
      // index cannot be read from the list alone. Those tables are read by the browser tier.
      const conditional = /\.\.\.\(/.test(columnsText);
      if (!conditional && money.size > 0 && !money.has(at)) {
        bad.push(`${file}:${line(node)} ${label}: the unit head is column ${at + 1}, but the Thousands cells sit in column ${[...money].map((i) => i + 1).join(', ')}. The marker sits over a column with no amount`);
      }
    };
    // The indexes of the td cells that hold a Thousands figure, read from each row of the table.
    const moneyColumnsOf = (node) => {
      const cols = new Set();
      const visit = (n) => {
        if (ts.isJsxElement(n) && n.openingElement.tagName.getText() === 'tr') {
          // A row's cells are its td and th children, in order: a row header is a th, and it takes a column.
          n.children.filter((c) => ts.isJsxElement(c) && ['td', 'th'].includes(c.openingElement.tagName.getText()))
            .forEach((cell, i) => { if (containsThousands(cell)) cols.add(i); });
        }
        ts.forEachChild(n, visit);
      };
      visit(node);
      return cols;
    };
    const visit = (node) => {
      if (ts.isJsxElement(node) && node.openingElement.tagName.getText() === 'DataTable' && containsThousands(node)) {
        const props = node.openingElement.attributes.properties;
        const attr = props.find((p) => ts.isJsxAttribute(p) && p.name.getText() === 'columns');
        const expr = attr && attr.initializer && ts.isJsxExpression(attr.initializer) ? attr.initializer.expression : null;
        const label = props.find((p) => ts.isJsxAttribute(p) && p.name.getText() === 'label');
        // A `columns` identifier is read through its const, so the fallback counts the marker in the array itself.
        const resolved = expr && ts.isIdentifier(expr) && consts.has(expr.text) ? consts.get(expr.text) : expr;
        // LOOSENING. The Precon campus table builds its columns in a same-file helper, `campusColumns(side)`, then slices it and
        // adds a head. The head reader cannot follow a slice or an index into a call, so the fallback counts the unit marks in
        // the columns text. That text holds only the call, so the fallback also reads the body of each same-file const the
        // columns call. The outcome it keeps: exactly ONE unit head in such a table. It no longer checks that the head sits
        // over the money column, for a table read this way. The browser tier reads that table's rendered head.
        const called = (text) => [text, ...[...new Set([...text.matchAll(/\b([A-Za-z_$][\w$]*)\(/g)].map((m) => m[1]))]
          .filter((name) => consts.has(name)).map((name) => consts.get(name).getText())].join('\n');
        check(node, `DataTable ${label && label.initializer ? label.initializer.getText() : ''}`, expr ? headsOf(expr) : [null], resolved ? called(resolved.getText()) : '');
      }
      if (ts.isJsxElement(node) && node.openingElement.tagName.getText() === 'table' && containsThousands(node)) {
        const heads = [];
        const collectTh = (n) => {
          if (ts.isJsxElement(n) && n.openingElement.tagName.getText() === 'th') heads.push(n.getText());
          ts.forEachChild(n, collectTh);
        };
        collectTh(node);
        check(node, 'table', heads.length > 0 ? heads : [null], node.getText());
      }
      ts.forEachChild(node, visit);
    };
    visit(sf);
  }
  assert.deepEqual(bad, [], `money tables without exactly one unit head:\n  ${bad.join('\n  ')}`);
});

// The Compare codes table (the owner, 2026-10-12: "let it scroll away"). One unit head, on its first money column.
// The per-job columns scroll away, so none of them carries a unit. Returns the problems, empty when it holds.
const unitRuleForCompareCodes = (src) => {
  const start = src.indexOf('const codeColumns = [');
  if (start < 0) return ['the Compare codes column list (codeColumns) was not found'];
  const block = src.slice(start, src.indexOf('\n  ];', start));
  const jobs = block.indexOf('comparison.labels.map');
  if (jobs < 0) return ['the Compare codes job columns (comparison.labels.map) were not found'];
  const units = (text) => (text.match(/moneyHead\(|\(in thousands\)|IN_THOUSANDS|\(K\)/g) ?? []).length;
  const problems = [];
  if (units(block.slice(0, jobs)) !== 1) problems.push(`the Compare codes table states its unit ${units(block.slice(0, jobs))} time(s) before its job columns, need one (its first money head)`);
  if (units(block.slice(jobs)) !== 0) problems.push('a scrolled job column of the Compare codes table states the unit, need none');
  return problems;
};
test('Compare codes states one unit, on its first money head, and no scrolled job column carries one', () => {
  const src = stripComments(readFileSync('src/components/screens/ComparisonScreen.tsx', 'utf8'), 'src/components/screens/ComparisonScreen.tsx');
  assert.deepEqual(unitRuleForCompareCodes(src), []);
});
test('the Compare codes unit rule goes red on a planted unit head in a job column', () => {
  const src = stripComments(readFileSync('src/components/screens/ComparisonScreen.tsx', 'utf8'), 'src/components/screens/ComparisonScreen.tsx');
  const planted = src.replace('<span className="min-w-0 truncate">{label}</span>', "<span className=\"min-w-0 truncate\">{label} {moneyHead('x')}</span>");
  assert.notEqual(planted, src, 'the plant did not land in the job column, so the red proof is void');
  assert.notDeepEqual(unitRuleForCompareCodes(planted), [], 'the rule does not catch a unit head in a scrolled job column');
});
